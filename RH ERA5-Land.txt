// -------------------- DATA --------------------

// Hourly ERA5-Land data for 2023.
// Temperature bands are in Kelvin.
var era5Hourly = ee.ImageCollection('ECMWF/ERA5_LAND/HOURLY')
  .filterDate('2023-01-01', '2024-01-01')
  .select(['temperature_2m', 'dewpoint_temperature_2m']);

// Saturation vapour pressure (Magnus formula), with temperatures in Celsius.
function addRelativeHumidity(image) {
  var temperatureC = image.select('temperature_2m').subtract(273.15);
  var dewpointC = image.select('dewpoint_temperature_2m').subtract(273.15);

  var es = temperatureC.expression(
    '6.112 * exp((17.67 * T) / (T + 243.5))',
    {T: temperatureC}
  );

  var e = dewpointC.expression(
    '6.112 * exp((17.67 * Td) / (Td + 243.5))',
    {Td: dewpointC}
  );

  var rh = e.divide(es).multiply(100).clamp(0, 100)
    .rename('RH');

  return rh.copyProperties(image, ['system:time_start']);
}

var rhHourly = era5Hourly.map(addRelativeHumidity);

// Daily means reduce chart volume while retaining a time series.
var dayCount = ee.Date('2024-01-01')
  .difference(ee.Date('2023-01-01'), 'day');

var days = ee.List.sequence(0, dayCount.subtract(1));

var rhDaily = ee.ImageCollection.fromImages(
  days.map(function(dayOffset) {
    var start = ee.Date('2023-01-01').advance(dayOffset, 'day');
    var dailyMean = rhHourly
      .filterDate(start, start.advance(1, 'day'))
      .mean()
      .set('system:time_start', start.millis());

    return dailyMean;
  })
);

// District boundaries (same asset as the supplied app).

// -------------------- DISTRICT DISPLAY --------------------

function display(distName) {
  var selected = ee.Feature(
    wbDist.filter(ee.Filter.eq('District', distName)).first()
  );
  var districtGeometry = selected.geometry();

  mapPanel.clear();

  mapPanel.addLayer(
    districtGeometry,
    {color: 'grey'},
    distName,
    true
  );

  // District daily mean RH time series.
  var chart = ui.Chart.image.series({
    imageCollection: rhDaily,
    region: districtGeometry,
    reducer: ee.Reducer.mean(),
    scale: 10000,
    xProperty: 'system:time_start'
  }).setOptions({
    title: 'Daily Relative Humidity for ' + distName + ' District',
    titleTextStyle: {bold: true, color: '04028b'},
    interpolateNulls: true,
    pointSize: 0,
    lineWidth: 1,
    hAxis: {
      title: 'Date',
      format: 'MMM',
      gridlines: {color: 'eeeeee'}
    },
    vAxis: {
      title: 'Relative Humidity (%)',
      viewWindow: {min: 0, max: 100},
      gridlines: {color: 'eeeeee'}
    },
    series: {
      0: {color: '0502ce'}
    },
    chartArea: {backgroundColor: 'EBEBEB'}
  });

  chartPanel.clear();
  chartPanel.add(chart);
  chartPanel.add(note);
  chartPanel.add(credit);
  chartPanel.add(dedication);

  // Annual mean RH map, clipped to the selected district.
  var annualMeanRH = rhHourly.mean().rename('RH');
  var rhVis = {
    min: 30,
    max: 100,
    palette: [
      '8c2d04', 'cc4c02', 'ec7014', 'fe9929',
      'fec44f', 'fff7bc', 'c7e9c0', '41b6c4',
      '2c7fb8', '253494'
    ]
  };

  mapPanel.addLayer(
    annualMeanRH.clip(districtGeometry),
    rhVis,
    'Annual Mean Relative Humidity (%)'
  );

  mapPanel.centerObject(districtGeometry);
  mapPanel.setOptions('HYBRID');

  // Map legend.
  var legend = ui.Panel({
    style: {
      position: 'bottom-right',
      backgroundColor: 'rgba(255,255,255,0.85)',
      padding: '8px'
    }
  });

  legend.add(ui.Label({
    value: 'Relative Humidity (%)',
    style: {fontWeight: 'bold', fontSize: '14px'}
  }));

  var colorBar = ui.Thumbnail({
    image: ee.Image.pixelLonLat().select('latitude')
      .visualize({
        min: -90,
        max: 90,
        palette: rhVis.palette
      }),
    params: {bbox: '0,0,10,100', dimensions: '20x160'},
    style: {height: '160px', width: '20px'}
  });

  var legendLabels = ui.Panel({
    widgets: [
      ui.Label('100%'),
      ui.Label(''),
      ui.Label(''),
      ui.Label(''),
      ui.Label('30%')
    ],
    style: {margin: '0 0 0 6px'}
  });

  legend.add(ui.Panel({
    widgets: [colorBar, legendLabels],
    layout: ui.Panel.Layout.flow('horizontal')
  }));

  mapPanel.add(legend);
}
