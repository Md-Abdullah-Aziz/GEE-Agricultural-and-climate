// ============================================================
// 1. USER SETTINGS
// ============================================================

var YEAR = 2025;

var startDate = ee.Date.fromYMD(YEAR, 1, 1);
var endDate = startDate.advance(1, 'year');

var CHIRPS_SCALE = 5566;

// ============================================================
// 2. USER INTERFACE
// ============================================================

var mapPanel = ui.Map();

var mainPanel = ui.Panel({
  style: {
    width: '500px',
    border: '3px solid #888'
  }
});

// Title
var title = ui.Label({
  value: 'Rainfall Explorer',
  style: {
    fontSize: '35px',
    fontFamily: 'serif',
    color: 'd10901',
    margin: '0 auto',
    padding: '10px 10px'
  }
});

// Subtitle
var subTitleEng = ui.Label({
  value: 'District-wise rainfall analysis using CHIRPS satellite data for the year ' + YEAR + '.',
  style: {
    fontSize: '15px',
    margin: '8px 0 8px 0',
    padding: '0 10px'
  }
});

// Data source
var dataSource = ui.Label({
  value: 'Data Source: CHIRPS Daily Precipitation - UCSB Climate Hazards Center',
  style: {
    fontSize: '13px',
    color: '0502ce',
    fontFamily: 'serif',
    margin: '8px 0 8px 0',
    padding: '0 20px'
  }
});

// Panels
var distPanel = ui.Panel();
var chartPanel = ui.Panel();

var bellowChartText = ui.Label({
  value: 'The chart represents the average daily rainfall (mm/day) across the selected district. The rainfall dataset is derived from satellite observations combined with ground-based information. Daily rainfall values are accumulated to generate the annual rainfall map.',
  style: {
    fontSize: '14px',
    padding: '0 10px'
  }
});

// Credits
var creditTitle = ui.Label({
  value: 'Developed using Google Earth Engine',
  style: {
    fontSize: '13px',
    color: 'c21301',
    fontFamily: 'serif',
    padding: '0 10px'
  }
});

// Add widgets
mainPanel.add(title);
mainPanel.add(subTitleEng);
mainPanel.add(dataSource);
mainPanel.add(distPanel);
mainPanel.add(chartPanel);

// Split panel
var splitPanel = ui.SplitPanel({
  firstPanel: mapPanel,
  secondPanel: mainPanel
});

ui.root.clear();
ui.root.add(splitPanel);


// ============================================================
// 3. DATASET: CHIRPS DAILY RAINFALL
// ============================================================

var chirps = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY')
  .filterDate(startDate, endDate)
  .select('precipitation');

// Annual rainfall accumulation (mm)
var annualRainfall = chirps.sum()
  .rename('Annual_Rainfall');

// Monthly rainfall collection
var monthlyRainfall = ee.ImageCollection(
  ee.List.sequence(1, 12).map(function(month) {

    var monthly = chirps.filter(
      ee.Filter.calendarRange(month, month, 'month')
    ).sum();

    return monthly.set({
      'month': month,
      'system:time_start': ee.Date.fromYMD(YEAR, month, 1).millis()
    });

  })
);


// ============================================================
// 4. DISTRICT BOUNDARIES
// ============================================================

// Use your original district shapefile asset
var wbDist = ee.FeatureCollection(
  'projects/ee-project-tapas/assets/wbDist'
);

var distNames = wbDist.aggregate_array('District');

print('District names:', distNames);


// ============================================================
// 5. DISTRICT SELECTION AND ANALYSIS
// ============================================================

var display = function(distName) {

  var selected = ee.Feature(
    wbDist.filter(
      ee.Filter.eq('District', distName)
    ).first()
  );

  var distGeometry = selected.geometry();

  mapPanel.clear();
  chartPanel.clear();

  // ----------------------------------------------------------
  // District boundary
  // ----------------------------------------------------------

  mapPanel.addLayer(
    selected,
    {color: 'red'},
    distName + ' Boundary'
  );


  // ==========================================================
  // 6. DISTRICT-WISE DAILY RAINFALL TIME SERIES
  // ==========================================================

  var rainfallChart = ui.Chart.image.series({

    imageCollection: chirps,

    region: distGeometry,

    reducer: ee.Reducer.mean(),

    scale: CHIRPS_SCALE,

    xProperty: 'system:time_start'

  })

  .setOptions({

    interpolateNulls: true,

    pointSize: 1.3,

    title: 'Daily Rainfall Analysis for ' +
           distName + ' District',

    titleTextStyle: {
      bold: true,
      color: '04028b'
    },

    hAxis: {
      format: 'MMM',
      title: '',
      gridlines: {
        color: 'ffffdd',
        count: 12
      }
    },

    vAxis: {

      title: 'Rainfall (mm/day)',

      titleTextStyle: {
        italic: false,
        color: '04028b'
      },

      gridlines: {
        color: 'ffffcc'
      },

      baselineColor: 'ffffcc'
    },

    series: {
      0: {
        lineWidth: 0.8,
        color: '0502ce',
        pointShape: 'circle',
        pointSize: 1.5
      }
    },

    chartArea: {
      backgroundColor: 'EBEBEB'
    }

  });

  chartPanel.add(rainfallChart);
  chartPanel.add(bellowChartText);


  // ==========================================================
  // 7. ANNUAL RAINFALL MAP
  // ==========================================================

  var meanRainfall = annualRainfall.clip(distGeometry);

  // Calculate minimum and maximum rainfall
  var stats = annualRainfall.reduceRegion({

    reducer: ee.Reducer.minMax(),

    geometry: distGeometry,

    scale: CHIRPS_SCALE,

    maxPixels: 1e10,

    bestEffort: true

  });


  // Rainfall visualization palette
  var rainfallPalette = [

    'ffffff',
    'd4f0ff',
    '92d5f7',
    '42b6e8',
    '168bd2',
    '0755a5',
    '08306b',
    '542788',
    '8073ac',
    'b2abd2'

  ];


  stats.evaluate(function(statsDict) {

    if (!statsDict ||
        statsDict.Annual_Rainfall_min === null ||
        statsDict.Annual_Rainfall_max === null) {

      print('Rainfall statistics unavailable for:', distName);
      return;

    }

    var min = statsDict.Annual_Rainfall_min;
    var max = statsDict.Annual_Rainfall_max;

    var rainfallVis = {

      min: min,

      max: max,

      palette: rainfallPalette

    };


    // Add annual rainfall layer
    mapPanel.addLayer(

      meanRainfall,

      rainfallVis,

      'Annual Rainfall (mm)'

    );


    mapPanel.centerObject(distGeometry);


    // ========================================================
    // 8. MAP LEGEND
    // ========================================================

    var legend = ui.Panel({

      style: {

        position: 'bottom-right',

        backgroundColor: 'rgba(255,255,255,0.85)'

      }

    });


    var legendTitle1 = ui.Label({

      value: 'Rainfall',

      style: {

        color: '#2d2d2d',

        fontWeight: 'bold',

        fontSize: '18px',

        margin: '5px auto'

      }

    });


    var legendTitle2 = ui.Label({

      value: '(Annual Accumulation)',

      style: {

        color: 'gray',

        fontSize: '13px',

        margin: '0 auto'

      }

    });


    legend.add(legendTitle1);
    legend.add(legendTitle2);


    // Create gradient
    var lon = ee.Image.pixelLonLat().select('latitude');

    var gradient = lon.multiply((max - min) / 100.0)
      .add(min);

    var legendImage = gradient.visualize(rainfallVis);


    var topRain = ui.Panel({

      widgets: [

        ui.Label({

          value: 'Max: ' + max.toFixed(1) + ' mm',

          style: {

            color: '08306b',

            textAlign: 'center',

            padding: '0 8px'

          }

        })

      ]

    });


    var thumbnail = ui.Thumbnail({

      image: legendImage,

      params: {

        bbox: '0,0,10,100',

        dimensions: '30x200'

      },

      style: {

        textAlign: 'center',

        maxWidth: '150px',

        margin: '0 auto',

        width: '30%'

      }

    });


    var bottRain = ui.Panel({

      widgets: [

        ui.Label({

          value: 'Min: ' + min.toFixed(1) + ' mm',

          style: {

            color: '168bd2',

            textAlign: 'center',

            padding: '0 8px'

          }

        })

      ]

    });


    legend.add(topRain);
    legend.add(thumbnail);
    legend.add(bottRain);

    mapPanel.add(legend);

    mapPanel.setOptions('HYBRID');

  });


  // ==========================================================
  // 9. DAILY RAINFALL ANIMATION
  // ==========================================================

  var rainfallAnimation = chirps.map(function(img) {

    return img.clip(distGeometry);

  });


  var animationParams = {

    bands: ['precipitation'],

    min: 0,

    max: 50,

    palette: rainfallPalette,

    region: distGeometry,

    dimensions: 450,

    framesPerSecond: 5

  };


  var animation = ui.Thumbnail({

    image: rainfallAnimation,

    params: animationParams,

    style: {

      position: 'top-left',

      margin: '0 auto'

    }

  });


  chartPanel.add(ui.Label({

    value: 'Daily Rainfall Animation',

    style: {

      fontSize: '16px',

      fontWeight: 'bold',

      color: '04028b',

      padding: '8px 10px'

    }

  }));


  chartPanel.add(animation);

  chartPanel.add(creditTitle);

};


// ============================================================

// ============================================================

distNames.evaluate(function(names) {

  var distDropDown = ui.Select({

    placeholder: 'Select a District / জেলা',

    items: names,

    onChange: display,

    style: {

      color: '053061',

      margin: '0 auto',

      width: '88%'

    }

  });

  distPanel.add(distDropDown);

});
